country_event = {
	id = SAF.20
	hidden = yes

	is_triggered_only = yes

	option = {
		SAF_refresh_cabinet_gap = yes
		SAF_refresh_cabinet_spirit = yes

		if = {
			limit = {
				has_country_flag = SAF_cabinet_open
				NOT = { has_completed_focus = SAF_the_walk_out }
				OR = {
					check_variable = { SAF_cabinet_gap > 30 }
					check_variable = { SAF_cabinet_gap < -30 }
				}
			}
			SAF_begin_walk_out_countdown = yes
		}
		else = {
			SAF_end_walk_out_countdown = yes
		}

		if = {
			limit = {
				has_country_flag = SAF_cabinet_open
				OR = {
					check_variable = { SAF_cabinet_spirit_now = 1 }
					check_variable = { SAF_cabinet_spirit_now = 5 }
				}
			}
			country_event = { id = SAF.20 days = 1 }
		}
		else = {
			clr_country_flag = SAF_walk_out_watch
		}
	}
}
