SAF_weekly_cabinet_drift = {
	if = {
		limit = { NOT = { has_country_flag = SAF_coalition_resolved } }

		set_variable = { SAF_dc_drift = SAF_gross_impact_dem }
		subtract_from_variable = { SAF_dc_drift = SAF_gross_impact_non }

		set_variable = { SAF_dc_non = party_popularity_100@neutrality }
		set_variable = { SAF_dc_dem = party_popularity_100@democratic }
		set_variable = { SAF_dc_fas = party_popularity_100@fascism }

		if = {
			limit = {
				check_variable = { SAF_dc_drift > 0 }
				check_variable = { var = SAF_dc_drift value = SAF_dc_non compare = greater_than }
			}
			set_variable = { SAF_dc_drift = SAF_dc_non }
		}
		else_if = {
			limit = { check_variable = { SAF_dc_drift < 0 } }
			set_variable = { SAF_dc_room = SAF_dc_dem }
			multiply_variable = { SAF_dc_room = -1 }
			if = {
				limit = { check_variable = { var = SAF_dc_drift value = SAF_dc_room compare = less_than } }
				set_variable = { SAF_dc_drift = SAF_dc_room }
			}
		}

		if = {
			limit = { NOT = { check_variable = { SAF_dc_drift = 0 } } }

			add_to_variable = { SAF_dc_dem = SAF_dc_drift }
			subtract_from_variable = { SAF_dc_non = SAF_dc_drift }

			set_popularities = {
				democratic = SAF_dc_dem
				neutrality = SAF_dc_non
				fascism = SAF_dc_fas
			}
		}
	}
}

			limit = { check_variable = { SAF_cabinet_gap > 10 } }

			limit = { check_variable = { SAF_cabinet_gap > -10 } }

		}
		else = {
			set_temp_variable = { SAF_spirit_target = 5 }
		}
		if = {
			limit = {
				NOT = { check_variable = { var = SAF_cabinet_spirit_now value = SAF_spirit_target compare = equals } }

				set_variable = { SAF_cab_conscription_factor = -0.10 }
			}
			if = {
				limit = { check_variable = { SAF_spirit_target = 2 } }
				set_variable = { SAF_cab_stability_factor = 0.05 }
				set_variable = { SAF_cab_political_power_factor = -0.10 }
				set_variable = { SAF_cab_war_support_factor = -0.05 }
			}
			if = {
				limit = { check_variable = { SAF_spirit_target = 3 } }
				set_variable = { SAF_cab_political_power_factor = -0.15 }
			}
			if = {
				limit = { check_variable = { SAF_spirit_target = 4 } }
				set_variable = { SAF_cab_war_support_factor = 0.05 }
				set_variable = { SAF_cab_political_power_factor = -0.10 }
				set_variable = { SAF_cab_stability_factor = -0.05 }
			}
			if = {
				limit = { check_variable = { SAF_spirit_target = 5 } }

				set_variable = { SAF_cab_stability_factor = -0.10 }

		}

	}
}

	set_variable = { SAF_smd_goods = SAF_smuts_debuff_level }
	multiply_variable = { SAF_smd_goods = SAF_smd_inc_goods }
	set_variable = { SAF_smd_mob = SAF_smuts_debuff_level }
	multiply_variable = { SAF_smd_mob = SAF_smd_inc_mob }

	set_temp_variable = { SAF_prev = SAF_smuts_debuff_level }
	add_to_temp_variable = { SAF_prev = 1 }
	clamp_temp_variable = { var = SAF_prev min = 0 max = 8 }
	subtract_from_temp_variable = { SAF_prev = SAF_smuts_debuff_level }
	set_variable = { SAF_smd_up_goods = SAF_prev }
	multiply_variable = { SAF_smd_up_goods = SAF_smd_inc_goods }
	set_variable = { SAF_smd_up_mob = SAF_prev }
	multiply_variable = { SAF_smd_up_mob = SAF_smd_inc_mob }

	set_temp_variable = { SAF_prev = SAF_smuts_debuff_level }
	add_to_temp_variable = { SAF_prev = -2 }
	clamp_temp_variable = { var = SAF_prev min = 0 max = 8 }
	subtract_from_temp_variable = { SAF_prev = SAF_smuts_debuff_level }
	set_variable = { SAF_smd_dn_goods = SAF_prev }
	multiply_variable = { SAF_smd_dn_goods = SAF_smd_inc_goods }
	set_variable = { SAF_smd_dn_mob = SAF_prev }
	multiply_variable = { SAF_smd_dn_mob = SAF_smd_inc_mob }

	SAF_sync_radicalised_cabinet_dm = yes

	force_update_dynamic_modifier = yes
}

SAF_hertzog_debuff_up = {

SAF_convert_one_civ_to_mil = {
	random_owned_state = {
		limit = {
			is_core_of = ROOT
			industrial_complex > 0
		}
		remove_building = {
			type = industrial_complex
			level = 1
		}
		add_building_construction = {
			type = arms_factory
			level = 1
			instant_build = yes
		}
	}
}

SAF_gsa_pay_one_civ = {
	if = {
		limit = { check_variable = { SAF_gsa_offmap_civs > 0 } }
		subtract_from_variable = { SAF_gsa_offmap_civs = 1 }
		add_offsite_building = { type = industrial_complex level = -1 }
	}
	else = {
		random_owned_state = {
			limit = { industrial_complex > 0 }
			remove_building = {
				type = industrial_complex
				level = 1
			}
		}
	}
	ENG = {
		add_offsite_building = { type = industrial_complex level = 1 }
	}
}
