SAF_mines_minister_seated = {
	custom_trigger_tooltip = {
		tooltip = SAF_minister_mines_seated_tt
		OR = {
			has_idea = SAF_sir_patrick_duncan
			has_idea = SAF_richard_stuttaford
		}
	}
}

SAF_cabinet_favours_hertzog = {
	check_variable = {
		var = party_popularity@neutrality
		value = party_popularity@democratic
		compare = greater_than
	}
}

SAF_cabinet_favours_smuts = {
	check_variable = {
		var = party_popularity@democratic
		value = party_popularity@neutrality
		compare = greater_than
	}
}

SAF_hertzog_wing_still_open = {
	OR = {
		custom_trigger_tooltip = {
			tooltip = SAF_coalition_open_tt
			NOT = { has_country_flag = SAF_coalition_resolved }
		}
		custom_trigger_tooltip = {
			tooltip = SAF_hertzog_wing_alone_tt
			has_country_flag = SAF_walk_out_hertzog_won
		}
	}
}
